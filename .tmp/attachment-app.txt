93:/* Fallback canvas line renderer — used only if lightweight-charts CDN fails. */
94:function fallbackLine(containerId, points, color) {
95:  const wrap = $(containerId).parentElement;
96:  let cv = wrap.querySelector("canvas.fbk");
97:  if (!cv) {
98:    cv = document.createElement("canvas");
99:    cv.className = "fbk";
100:    cv.style.cssText = "width:100%;height:100%;display:block";
101:    wrap.appendChild(cv);
102:    $(containerId).style.display = "none";
103:  }
104:  const dpr = window.devicePixelRatio || 1;
105:  const W = wrap.clientWidth, H = wrap.clientHeight;
106:  cv.width = W * dpr; cv.height = H * dpr;
107:  const g = cv.getContext("2d");
108:  g.scale(dpr, dpr);
109:  g.clearRect(0, 0, W, H);
110:  if (!points.length) return;
111:  const vs = points.map((p) => p.v ?? p.c);
112:  const lo = Math.min(...vs), hi = Math.max(...vs), pad = (hi - lo) * 0.08 || 1;
113:  const X = (i) => 8 + (i / Math.max(points.length - 1, 1)) * (W - 60);
114:  const Y = (v) => 14 + (1 - (v - lo + pad) / (hi - lo + 2 * pad)) * (H - 40);
115:  g.strokeStyle = "#e7e7e7"; g.lineWidth = 1;
116:  for (let i = 0; i <= 4; i++) {
117:    const y = 14 + (i / 4) * (H - 40);
118:    g.beginPath(); g.moveTo(8, y); g.lineTo(W - 52, y); g.stroke();
119:    g.fillStyle = "#9ca3af"; g.font = "10px Inter,sans-serif";
120:    g.fillText(fmtBig(hi - (i / 4) * (hi - lo)), W - 48, y + 3);
121:  }
122:  g.strokeStyle = color || "#E30613"; g.lineWidth = 2; g.beginPath();
123:  points.forEach((p, i) => { const v = p.v ?? p.c; i ? g.lineTo(X(i), Y(v)) : g.moveTo(X(i), Y(v)); });
124:  g.stroke();
125:  usedFallbackRenderer = true;
126:}
